/* ================= LICENCE ================= */
const INSTALL_CODE="CM-INSTALL-982314";
const ACTIVATE_CODE="CM-ACTIVE-2026";
const DAYS_VALID=180;
const ALERT_DAYS=10;

function deviceId(){
 return btoa(navigator.userAgent+screen.width+screen.height);
}

function showLock(msg){
 lockScreen.classList.remove("hidden");
 app.classList.add("hidden");
 lockMsg.innerText=msg;
}

function checkApp(){
 const lic=JSON.parse(localStorage.getItem("license"));
 const now=Date.now();
 if(!lic){showLock("Code d’installation requis");return;}
 if(lic.device!==deviceId()){showLock("Licence invalide");return;}
 if(now<lic.last){showLock("Date système modifiée");return;}
 const days=Math.ceil((lic.expire-now)/86400000);
 if(days<=ALERT_DAYS){
  alertExpire.innerText="⚠️ Expire dans "+days+" jours";
  alertExpire.classList.remove("hidden");
 }
 if(now>lic.expire){showLock("Licence expirée");return;}
 lic.last=now;
 localStorage.setItem("license",JSON.stringify(lic));
 lockScreen.classList.add("hidden");
 app.classList.remove("hidden");
}

function checkLicense(){
 const c=licenseCode.value.trim();
 const now=Date.now();
 if(c===INSTALL_CODE){
  localStorage.setItem("license",JSON.stringify({
   device:deviceId(),
   expire:now+DAYS_VALID*86400000,
   last:now
  }));
  location.reload();
 }
 if(c===ACTIVATE_CODE){
  const l=JSON.parse(localStorage.getItem("license"));
  l.expire=now+DAYS_VALID*86400000;
  l.last=now;
  localStorage.setItem("license",JSON.stringify(l));
  location.reload();
 }
}
