const INSTALL_CODE="CM-INSTALL-982314";
const ACTIVATION_CODE="CM-ACTIVE-2026-01";
const ACTIVATION_DAYS=180;
const ALERT_DAYS=10;

/* ===== FORMAT HTG ===== */
const formatHTG=v=>new Intl.NumberFormat("fr-HT",{style:"currency",currency:"HTG"}).format(v);

/* ===== LICENCE ===== */
function deviceId(){return btoa(navigator.userAgent+screen.width);}
function checkAppLock(){
 const lic=JSON.parse(localStorage.getItem("license"));
 const now=Date.now();
 if(!lic){showLock("Code requis");return false;}
 if(now<lic.last){showLock("Date système modifiée");return false;}
 if(lic.device!==deviceId()){showLock("Licence invalide");return false;}
 const days=Math.ceil((lic.expire-now)/86400000);
 if(days<=ALERT_DAYS){alertExpire.innerText="⚠️ Expire dans "+days+" jours";alertExpire.classList.remove("hidden")}
 if(now>lic.expire){showLock("Activation expirée");return false;}
 lic.last=now;localStorage.setItem("license",JSON.stringify(lic));return true;
}
function showLock(m){document.body.innerHTML="";document.body.appendChild(lockScreen);lockScreen.classList.remove("hidden");lockMsg.innerText=m}
function checkLicense(){
 const c=licenseCode.value.trim(),now=Date.now();
 if(c===INSTALL_CODE){
  localStorage.setItem("license",JSON.stringify({device:deviceId(),expire:now+ACTIVATION_DAYS*86400000,last:now}));
  location.reload()
 }
 if(c===ACTIVATION_CODE){
  const l=JSON.parse(localStorage.getItem("license"));
  l.expire=now+ACTIVATION_DAYS*86400000;l.last=now;
  localStorage.setItem("license",JSON.stringify(l));location.reload()
 }
}
if(!checkAppLock()) throw "";

function printDaily(){
 const today=new Date().toLocaleDateString();
 reportDate.innerText=today;
 let html="",tot={Moncash:0,Natcash:0};
 store.data.history.filter(x=>x.date===today).forEach(x=>{
  tot[x.service]+=x.type==="depot"?x.amount:-x.amount;
 });
 html+=`Moncash : ${formatHTG(tot.Moncash)}<br>`;

 if(t==="retrait"&&a>store.data[s]){alert("Solde insuffisant");return;}

}
